  <script>
    let base64ImageData = null;

    // Load saved API Key on start
    window.addEventListener('DOMContentLoaded', () => {
      const savedKey = localStorage.getItem('gemini_api_key');
      if (savedKey) {
        updateApiStatus(true);
      } else {
        updateApiStatus(false);
      }
    });

    function updateApiStatus(isConfigured) {
      const statusText = document.getElementById('api-status-text');
      if (isConfigured) {
        statusText.innerText = 'API Key 已就緒';
        statusText.classList.add('text-yellow-200');
      } else {
        statusText.innerText = '未設定 Key';
      }
    }

    function openApiKeyModal() {
      const savedKey = localStorage.getItem('gemini_api_key') || '';
      document.getElementById('api-key-input').value = savedKey;
      document.getElementById('api-modal').classList.remove('hidden');
    }

    function closeApiKeyModal() {
      document.getElementById('api-modal').classList.add('hidden');
    }

    function saveApiKey() {
      const key = document.getElementById('api-key-input').value.trim();
      if (key) {
        localStorage.setItem('gemini_api_key', key);
        updateApiStatus(true);
        closeApiKeyModal();
      } else {
        alert('請輸入有效的 API Key');
      }
    }

    function handleImageSelect(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        base64ImageData = e.target.result.split(',')[1];
        document.getElementById('image-preview').src = e.target.result;
        document.getElementById('upload-prompt').classList.add('hidden');
        document.getElementById('preview-container').classList.remove('hidden');
        document.getElementById('analyze-btn').disabled = false;
      };
      reader.readAsDataURL(file);
    }

    function removeImage(event) {
      event.stopPropagation();
      base64ImageData = null;
      document.getElementById('image-input').value = '';
      document.getElementById('preview-container').classList.add('hidden');
      document.getElementById('upload-prompt').classList.remove('hidden');
      document.getElementById('analyze-btn').disabled = true;
    }

    function resetApp() {
      removeImage(new Event('click'));
      document.getElementById('result-box').classList.add('hidden');
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    async function startAnalysis() {
      const apiKey = localStorage.getItem('gemini_api_key');
      if (!apiKey) {
        openApiKeyModal();
        return;
      }

      if (!base64ImageData) {
        alert('請先上傳學生手寫相片！');
        return;
      }

      // UI Switch
      document.getElementById('loading-box').classList.remove('hidden');
      document.getElementById('result-box').classList.add('hidden');
      document.getElementById('analyze-btn').disabled = true;

      const promptText = `
你是香港小學四年級資深中文科老師。請分析學生上傳的手寫造句圖片，並嚴格以 JSON 格式回應，格式如下：

{
  "ocr_text": "辨識到的學生手寫完整原句（繁體中文）",
  "corrected_sentence": "修正錯別字及語病後的標準原句",
  "error_diagnosis": "詳細分析句中的錯別字（指出錯在哪裡）、語病或詞義不當之處。若無錯誤請給予讚賞。",
  "enrichment_vivid": "小四級生動好詞增潤版（加入感官描寫、修飾詞或動態形容）",
  "enrichment_rhetoric": "小四級修辭升級版（運用小四課程常見修辭如比喻、擬人、排比或關聯詞，如『雖然……但是……』、『像……一樣』）",
  "teacher_feedback": "一段親切、鼓勵性的老師評語，適合小四學生閱讀。"
}

請確保回應內容符合香港教育局小學四年級中文科課程要求，使用標準繁體中文。請只輸出 JSON 本身，不要添加任何 Markdown 標記或 extra text。
`;

      try {
        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{
              parts: [
                { text: promptText },
                { inline_data: { mime_type: "image/jpeg", data: base64ImageData } }
              ]
            }]
          })
        });

        const data = await response.json();
        
        if (data.error) {
          throw new Error(data.error.message || 'API 呼叫失敗');
        }

        const candidateText = data.candidates[0].content.parts[0].text;
        // Clean markdown codeblocks if present
        const jsonString = candidateText.replace(/```json/g, '').replace(/```/g, '').trim();
        const result = JSON.parse(jsonString);

        // Display results
        document.getElementById('ocr-result-text').innerText = result.ocr_text || '未能辨識字詞';
        document.getElementById('corrected-sentence').innerText = result.corrected_sentence || '--';
        document.getElementById('error-diagnosis').innerText = result.error_diagnosis || '--';
        document.getElementById('enrichment-vivid').innerText = result.enrichment_vivid || '--';
        document.getElementById('enrichment-rhetoric').innerText = result.enrichment_rhetoric || '--';
        document.getElementById('teacher-feedback').innerText = result.teacher_feedback || '--';

        document.getElementById('loading-box').classList.add('hidden');
        document.getElementById('result-box').classList.remove('hidden');
        document.getElementById('result-box').scrollIntoView({ behavior: 'smooth' });

      } catch (err) {
        alert('分析失敗：' + err.message);
        document.getElementById('loading-box').classList.add('hidden');
      } finally {
        document.getElementById('analyze-btn').disabled = false;
      }
    }
  </script>
